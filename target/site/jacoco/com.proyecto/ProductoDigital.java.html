package com.proyecto;


/**
 * Subclase de Producto que representa un producto digital
 * (software, licencia, eBook, etc.).
 *
 * Implementa calcularPrecioFinal() aplicando un descuento del 10%
 * sobre el precio base, simulando la ausencia de IVA o costes de distribución.
 */
public class ProductoDigital extends Producto {

    // ── Atributo propio ────────────────────────────────────────────────────
    private double tamañoMB;   // tamaño de descarga en megabytes

    // ── Constructor ────────────────────────────────────────────────────────
    public ProductoDigital(String nombre, double precio, double tamañoMB) {
        super(nombre, precio);
        this.tamañoMB = tamañoMB;
    }

    // ── Getters y Setters ──────────────────────────────────────────────────
    public double getTamañoMB() {
        return tamañoMB;
    }

    public void setTamañoMB(double tamañoMB) {
        this.tamañoMB = tamañoMB;
    }

    // ── calcularPrecioFinal ────────────────────────────────────────────────
    /**
     * Precio final = precio base * 0.90 (descuento del 10%).
     *
     * @return precio con descuento aplicado
     */
    @Override
    public double calcularPrecioFinal() {
        return precio * 0.90;
    }

    // ── toString ───────────────────────────────────────────────────────────
    @Override
    public String toString() {
        return "ProductoDigital{"
                + "nombre='" + nombre + "'"
                + ", precio=" + precio
                + ", tamañoMB=" + tamañoMB
                + ", precioFinal=" + calcularPrecioFinal()
                + "}";
    }
}
